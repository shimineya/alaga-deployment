-- 1. USERS TABLE
-- Stores accounts for Medical Staff and Caregivers
CREATE TABLE users (
    user_id SERIAL PRIMARY KEY,
    username VARCHAR(50) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    email VARCHAR(100) NOT NULL UNIQUE,
    first_name VARCHAR(50),
    last_name VARCHAR(50),
    role VARCHAR(20) NOT NULL, -- 'Medical Staff', 'Caregiver', 'Admin'
    is_biometric_enabled BOOLEAN DEFAULT FALSE,
    is_mfa_enabled BOOLEAN DEFAULT FALSE,
    mfa_secret VARCHAR(255),
    mfa_method VARCHAR(20),
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

-- 2. PATIENTS TABLE
-- Stores Infant/Patient profiles (linked to sensors via MAC address)
CREATE TABLE patients (
    patient_id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    birthdate DATE NOT NULL,
    patient_type VARCHAR(50), -- e.g., 'Infant', 'Bedridden'
    device_mac_address VARCHAR(50) UNIQUE, -- The ID of the ESP32
    baseline_data JSONB, -- Stores "normal" ranges for the One-Class SVM
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

-- 3. PATIENT_ACCESS TABLE
-- Links Users to Patients (Many-to-Many relationship)
CREATE TABLE patient_access (
    access_id SERIAL PRIMARY KEY,
    user_id INT REFERENCES users(user_id) ON DELETE CASCADE,
    patient_id INT REFERENCES patients(patient_id) ON DELETE CASCADE,
    relationship VARCHAR(50), -- e.g., 'Parent', 'Nurse'
    access_level VARCHAR(20) DEFAULT 'View', 
    assigned_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

-- 4. SENSOR_READINGS TABLE
-- High-volume table for ESP32 data (Vital Signs + Moisture)
CREATE TABLE sensor_readings (
    reading_id BIGSERIAL PRIMARY KEY,
    patient_id INT REFERENCES patients(patient_id) ON DELETE CASCADE,
    heart_rate INT,
    spo2 INT,
    temperature DECIMAL(5,2),
    moisture_value INT,
    recorded_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

-- 5. ANOMALY_EVENTS TABLE
-- Stores AI detections (One-Class SVM output)
CREATE TABLE anomaly_events (
    event_id SERIAL PRIMARY KEY,
    patient_id INT REFERENCES patients(patient_id) ON DELETE CASCADE,
    reading_id BIGINT REFERENCES sensor_readings(reading_id),
    anomaly_type VARCHAR(50), -- 'Bed-wetting', 'Abnormal Vital Sign'
    ocsvm_score DECIMAL(10,4), -- The AI confidence score
    detected_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

-- 6. ALERT_NOTIFICATIONS TABLE
-- Log of alerts sent to the mobile app
CREATE TABLE alert_notifications (
    alert_id SERIAL PRIMARY KEY,
    event_id INT REFERENCES anomaly_events(event_id),
    user_id INT REFERENCES users(user_id),
    status VARCHAR(20) DEFAULT 'Sent',
    message TEXT,
    sent_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

-- 7. REPORTS TABLE
-- Stores references to generated PDF/Excel reports
CREATE TABLE reports (
    report_id SERIAL PRIMARY KEY,
    user_id INT REFERENCES users(user_id),
    patient_id INT REFERENCES patients(patient_id),
    report_type VARCHAR(50),
    file_url VARCHAR(255),
    generated_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

-- 8. ACCESS_LOGS TABLE
-- Security audit trail (Required for your Ethical Data Handling)
CREATE TABLE access_logs (
    log_id BIGSERIAL PRIMARY KEY,
    user_id INT REFERENCES users(user_id),
    target_patient_id INT REFERENCES patients(patient_id),
    action VARCHAR(255),
    timestamp TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

-- PERFORMANCE INDEX
-- Makes loading the dashboard charts faster
CREATE INDEX idx_readings_patient_time ON sensor_readings(patient_id, recorded_at DESC);
